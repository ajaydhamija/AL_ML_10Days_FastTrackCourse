\sect{4. Formula Sheets}
\begin{frame}{Key Formulas (I) --- Data, Supervised Learning \& Fusion (L1--L3)}
\begin{columns}[T]
\column{0.50\textwidth}
\begin{fbx}{Data \& statistics (L1)}\[\text{Error}=\text{Bias}^2+\text{Variance}+\varepsilon,\quad P(\ge1\text{ FP})=1-(1-\alpha)^m\]\[\text{PPV}=\frac{\text{sens}\cdot p}{\text{sens}\cdot p+(1-\text{spec})(1-p)}\]\[D_{KS}=\max_x|F_1-F_2|\]\end{fbx}
\begin{fbx}{Classification (L2)}\[P=\tfrac{TP}{TP+FP},\ R=\tfrac{TP}{TP+FN},\ F_\beta=(1+\beta^2)\tfrac{PR}{\beta^2P+R}\]\[G=1-\textstyle\sum p_k^2,\quad H=-\sum p_k\log_2p_k\]\[\text{margin}=2/\|w\|\]\[\min\tfrac12\|w\|^2+C\textstyle\sum\xi_i,\quad K=e^{-\gamma\|x-x'\|^2}\]\end{fbx}
\column{0.48\textwidth}
\begin{fbx}{Regression (L3)}\[\hat\beta=(X^{\top}X)^{-1}X^{\top}y,\quad\hat\beta_{\text{ridge}}=(X^{\top}X+\lambda I)^{-1}X^{\top}y\]\[\text{Lasso: }\min\|y-X\beta\|^2+\lambda\|\beta\|_1,\quad R^2=1-\tfrac{SS_{\text{res}}}{SS_{\text{tot}}}\]\end{fbx}
\begin{fbx}{Kalman filter (L3)}\[\hat x^{-}=F\hat x+Bu,\quad P^{-}=FPF^{\top}+Q\]\[K=P^{-}H^{\top}(HP^{-}H^{\top}+R)^{-1}\]\[\hat x=\hat x^{-}+K(z-H\hat x^{-}),\quad P=(I-KH)P^{-}\]\end{fbx}
\end{columns}
\end{frame}

\begin{frame}{Key Formulas (II) --- Unsupervised Learning \& Anomaly Detection (L4--L5)}
\begin{columns}[T]
\column{0.50\textwidth}
\begin{fbx}{Clustering (L4)}\[J=\sum_j\sum_{x\in C_j}\|x-\mu_j\|^2,\quad\mu_j=\tfrac{1}{|C_j|}\sum_{x\in C_j}x\]\[s(i)=\frac{b(i)-a(i)}{\max(a(i),b(i))}\in[-1,1]\]\[P(x)=\sum_k\pi_k\mathcal N(x;\mu_k,\Sigma_k),\quad r_{ik}=\tfrac{\pi_k\mathcal N_{ik}}{\sum_j\pi_j\mathcal N_{ij}}\]\[C=V\Lambda V^{\top},\quad Z=\bar XV_k,\quad\text{EVR}_j=\lambda_j/\textstyle\sum\lambda\]\end{fbx}
\sm{DBSCAN: core if $|N_\varepsilon(p)|\ge$ MinPts; clusters are maximal density-connected sets; ARI measures agreement between two clusterings.}
\column{0.48\textwidth}
\begin{fbx}{Anomaly detection (L5)}\[D_M(x)=\sqrt{(x-\mu)^{\top}\Sigma^{-1}(x-\mu)}\]\[s(x,\psi)=2^{-E[h(x)]/c(\psi)},\ c(\psi)=2H(\psi{-}1)-\tfrac{2(\psi-1)}{\psi}\]\[\text{RE}(x)=\|x-f_{\text{dec}}(f_{\text{enc}}(x))\|^2\]\[f(x)=\text{sign}\Big(\sum\alpha_iK(x_i,x)-\rho\Big)\]\end{fbx}
\begin{wbox}\scriptsize\textbf{Key pitfall:} the plain covariance $\Sigma$ in Mahalanobis is contaminated by the anomalies themselves --- use the MCD estimator (most central 50\%).\end{wbox}
\end{columns}
\end{frame}

\begin{frame}{Key Formulas (III) --- Reinforcement Learning \& MARL (L6--L8)}
\begin{columns}[T]
\column{0.33\textwidth}
\begin{fbx}{Q-learning (L6)}\[Q^{*}=\sum_{s'}P\big[R+\gamma\max_{a'}Q^{*}\big]\]\[Q\leftarrow Q+\alpha\delta\]\[\delta=r+\gamma\max Q'-Q\]\[y=r+\gamma Q(s',a^{*};\theta^{-})\]\[Q=V+A-\bar A\]\end{fbx}
\column{0.33\textwidth}
\begin{fbx}{Policy gradient (L7)}\[\nabla J=\mathbb E[\nabla\log\pi\,G_t]\]\[\hat A=r+\gamma V(s')-V(s)\]\[r_t=\pi_{\text{new}}/\pi_{\text{old}}\]\[L^{\text{CLIP}}=\mathbb E[\min(r_t\hat A,\text{clip}\,\hat A)]\]\[J_{\text{SAC}}=\mathbb E[r+\alpha H(\pi)]\]\end{fbx}
\column{0.32\textwidth}
\begin{fbx}{MARL (L8)}\[\partial Q_{\text{tot}}/\partial Q_i>0\]\[L=\tfrac1N\sum_iL_i^{\text{CLIP}}\]\[A_i=Q(s,\mathbf a)-b_i(s,\mathbf a_{-i})\]\[b_i=\mathbb E_{a_i'\sim\pi_i}Q(s,(a_i',\mathbf a_{-i}))\]\end{fbx}
\end{columns}
\begin{columns}[T]
\column{0.50\textwidth}
\begin{fbx}{$\varepsilon$-greedy and annealing}\[P(\text{greedy})=1-\varepsilon+\tfrac{\varepsilon}{|A|},\quad\varepsilon_t=\varepsilon_{\min}+(\varepsilon_{\max}-\varepsilon_{\min})e^{-t/\tau}\]\end{fbx}
\column{0.48\textwidth}
\begin{fbx}{Shaping and discounting}\[F=\gamma\Phi(s')-\Phi(s),\qquad G_t=\sum_k\gamma^kr_{t+k+1}\]\end{fbx}
\end{columns}
\end{frame}

\sect{5. Misconceptions}
\begin{frame}{Common Misconceptions --- Exam Traps and Conceptual Errors (I)}
\figc{5.0cm}{\begin{tikzpicture}
\tikzset{M/.style={nd,text width=7.0cm,align=left,minimum height=1.15cm,font=\fontsize{6.5}{7.7}\selectfont,inner sep=3pt,draw=ISSAred,fill=ISSAred!4}}
\node[M]at(-3.75,1.9){\textbf{``More data always helps.''} \red{False.} Biased data amplifies bias: more Afghan data would not fix Iraq (94.1\% $\to$ 58.7\%). Quality and domain match beat quantity.};
\node[M]at(-3.75,0.6){\textbf{``High accuracy = good model.''} \red{False.} On the real IADS stream (78.4\% civilian) ``always civilian'' scores 78\%+. Check per-class recall, AUC, confusion matrix.};
\node[M]at(-3.75,-0.7){\textbf{``Percentages can be added or averaged.''} \red{False.} Simpson: A wins both stone sizes yet loses 78\% vs 83\% overall --- totals come from counts.};
\node[M]at(-3.75,-2.0){\textbf{``Not significant = no effect.''} \red{False.} The coating trial ($p=0.072$, 40 sorties) was under-powered; 93 per arm were needed.};
\node[M]at(3.75,1.9){\textbf{``DBSCAN needs $k$.''} \red{False.} It discovers the number of clusters from density ($\varepsilon$, MinPts); $k$-Means and GMM need $k$.};
\node[M]at(3.75,0.6){\textbf{``A baseline biases the gradient.''} \red{False.} $\mathbb E[b(s)\nabla\log\pi]=0$ for any $b$ not depending on $a$: less variance, zero bias.};
\node[M]at(3.75,-0.7){\textbf{``Replay is only for efficiency.''} \red{Incomplete.} It also breaks temporal correlation --- without that, DQN often diverges.};
\node[M]at(3.75,-2.0){\textbf{``t-SNE distances are meaningful.''} \red{False.} Only local neighbourhoods are preserved; never compare between-cluster distances.};
\end{tikzpicture}}
\end{frame}

\begin{frame}{Common Misconceptions (II) --- More Traps}
\figc{4.4cm}{\begin{tikzpicture}
\tikzset{M/.style={nd,text width=7.0cm,align=left,minimum height=1.3cm,font=\fontsize{6.5}{7.7}\selectfont,inner sep=3pt,draw=ISSAred,fill=ISSAred!4}}
\node[M]at(-3.75,1.4){\textbf{``QMIX just sums Q-values.''} \red{Misleading.} VDN sums; QMIX mixes non-linearly, conditioned on the global state through a hypernetwork, while keeping monotonicity.};
\node[M]at(-3.75,0){\textbf{``SAC always needs more episodes than PPO.''} \red{Context-dependent.} SAC took more episodes to 70\% on the drone task (1,500 vs 800), but reuses data via replay; choose by whether compute or environment interaction is scarce.};
\node[M]at(-3.75,-1.4){\textbf{``Q = 18.7 means 18.7\% chance to win.''} \red{False.} Q-values are expected discounted reward in reward units, not probabilities.};
\node[M]at(3.75,1.4){\textbf{``A 99\%-accurate detector's alarms are 99\% reliable.''} \red{False.} At a 0.1\% base rate only $\approx$9\% of alarms are real threats.};
\node[M]at(3.75,0){\textbf{``Strong correlation shows the cause.''} \red{False.} Catering vs wins $r=0.71$ fell to 0.09 after controlling for budget; maintenance hours vs breakdowns is reverse causation.};
\node[M]at(3.75,-1.4){\textbf{``Simulation win rate = field performance.''} \red{False.} Drones 94\% $\to$ 78\% real without randomisation; a human OPFOR adapts.};
\end{tikzpicture}}
\begin{ibox}\scriptsize\textbf{Exam technique:} when a question offers a confident absolute (``always'', ``proves'', ``guarantees''), look for the hidden assumption --- base rate, sample size, confounder, distribution shift, or simulation.\end{ibox}
\end{frame}

\sect{7. Ethics}
\begin{frame}{Ethical AI in Defence --- Eight Non-Negotiable Principles}

\node[E]at(1.8,1.1){\textbf{3 Explainability.} SHAP for classifiers, per-feature RE for anomalies, Q-value advisories --- targeting needs a rationale.};
\node[E]at(5.4,1.1){\textbf{4 Proportionality.} IHL Art.\ 51(5)(b): no reward function can encode this legal, contextual judgement.};
\node[E]at(-5.4,-1.1){\textbf{5 Distinction.} Combatant vs civilian: novel uniforms, dual-use sites, surrender --- no AI solves this reliably at tempo.};
\node[E]at(-1.8,-1.1){\textbf{6 Adversarial robustness.} Fixed policies can be reverse-engineered; classifiers fooled; red-team before deployment.};
\node[E]at(1.8,-1.1){\textbf{7 Accountability.} Chain of command remains; full audit log: input $\to$ AI advice $\to$ human decision $\to$ outcome.};
\node[E]at(5.4,-1.1){\textbf{8 Sim-to-real honesty.} 82.4\% in simulation $\neq$ operational; drones 94\% $\to$ 78\%. Overstating AI is an ethical failure.};

\begin{wbox}\scriptsize\centering\textbf{Ethical bright line:} AI is a DECISION-SUPPORT tool. Human commanders retain authority for all lethal action.\end{wbox}
\end{frame}
